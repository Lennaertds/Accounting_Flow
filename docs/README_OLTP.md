@startuml
' --- LAYOUT OPTIMALISATIE ---
skinparam linetype ortho
skinparam packageStyle rectangle
skinparam shadowing false
skinparam nodesep 100
skinparam ranksep 80
skinparam padding 8

package "Thema 0: Foundation & Auth" #white {
    entity "organizations" as org {
        * id : uuid [PK]
        --
        * name : varchar(255)
        vat : varchar(50)
        iban : varchar(34)
        * created_at : timestamp
    }

    entity "users" as user {
        * id : uuid [PK]
        --
        * org_id : uuid [FK]
        * first_name : varchar(255)
        * last_name : varchar(255)
        * email : varchar(255) [UQ]
        * password_hash : text
        * role : varchar(20)
        last_login : timestamp
    }
}

package "Thema 1: Dimensions" #f3e5f5 {
    entity "partners" as partner {
        * id : uuid [PK]
        --
        * org_id : uuid [FK]
        * name : varchar(255)
        * is_peppol_ready : boolean
        * preferred_order_channel : varchar(20)
        * preferred_invoice_channel : varchar(20)
        vat_nr : varchar(50)
        peppol_id : varchar(100)
        email : varchar(255)
        * is_customer : boolean
        * is_supplier : boolean
        bank_account_iban : varchar(34)
    }

    entity "products" as prod {
        * id : uuid [PK]
        --
        * org_id : uuid [FK]
        sku : varchar(100)
        * name : varchar(255)
        unit : varchar(20)
        active : boolean
    }

    entity "vat_codes" as vat {
        * id : uuid [PK]
        --
        * org_id : uuid [FK]
        * code : text [UQ]
        * rate : decimal(5,2)
        description : text
        * tax_declaration_box : integer
    }

    entity "analytic_accounts" as analytic {
        * id : uuid [PK]
        --
        * org_id : uuid [FK]
        * name : varchar(100)
        * code : varchar(50) [UQ]
        * active : boolean
    }
}

package "Thema 2: Banking & Reconciliation" #E1F5FE {
    
    entity "bank_statements" as statements {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        date : date
        amount : decimal(12,2)
        communication : text
        counterparty_iban : varchar(34)
        bank_reference : varchar(255) <<UQ>>
    }

    entity "sepa_exports" as sepa {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        created_by : uuid [FK]
        created_at : timestamp
        filename : varchar(255)
        file_storage_path : varchar(512)
        total_amount : decimal(12,2)
        transaction_count : integer
        execution_date : date
        status : varchar(30) <<GENERATED...>>
        bank_batch_reference : varchar(100)
    }
}

package "Thema 3: Inkoop & Voorraad" #E8F5E9 {

    entity "quotations" as quot {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        partner_id : uuid [FK]
        quotation_nr : varchar
        version : integer
        status : varchar
    }

    entity "outgoing_orders" as out_ord {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        supplier_id : uuid [FK]
        order_nr : varchar
        status : varchar
    }

    entity "outgoing_order_lines" as out_ord_l {
        * id : uuid [PK]
        --
        order_id : uuid [FK]
        product_id : uuid [FK]
        quantity : decimal
        unit_price : decimal
    }

    entity "goods_receipts" as receipt {
        * id : uuid [PK]
        --
        outgoing_order_id : uuid [FK]
        received_by : uuid [FK]
        delivery_note_ref : varchar
    }

    entity "goods_receipt_lines" as receipt_l {
        * id : uuid [PK]
        --
        receipt_id : uuid [FK]
        product_id : uuid [FK]
        quantity_received : decimal
    }

    entity "incoming_invoices" as inc_inv {
        * id : uuid [PK]
        --
        supplier_id : uuid [FK]
        invoice_nr : varchar
        sepa_export_id : uuid [FK]
        status : varchar
    }

    entity "incoming_invoice_lines" as inc_inv_l {
        * id : uuid [PK]
        --
        invoice_id : uuid [FK]
        product_id : uuid [FK]
        ledger_account_id : uuid [FK]
    }

    entity "inventory_mutations" as inv_mut {
        * id : uuid [PK]
        --
        product_id : uuid [FK]
        source_type : varchar
        source_id : uuid
        change : decimal
    }
}

package "Thema 4: Verkoop" #FFF3E0 {

    entity "incoming_orders" as inc_ord {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        customer_id : uuid [FK]
        order_nr : varchar(100)
        date : date
        source : varchar(50)
        reference_ext : varchar(255) <<UQ>>
        total_estimated : decimal
        status : varchar(30)
        ai_summary : text
        file_hash : text
    }

    entity "incoming_order_lines" as inc_ord_l {
        * id : uuid [PK]
        --
        order_id : uuid [FK]
        product_id : uuid [FK]
        description : text
        quantity : decimal
        unit_price : decimal
        vat_code_id : uuid [FK]
    }

    entity "outgoing_invoices" as out_inv {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        customer_id : uuid [FK]
        invoice_nr : varchar(100) <<UQ>>
        date : date
        due_date : date
        currency : varchar(3)
        total_excl_vat : decimal
        vat_code_id : uuid [FK]
        status : varchar(30)
        peppol_status : varchar(30)
        file_hash : text
    }

    entity "outgoing_invoice_lines" as out_inv_l {
        * id : uuid [PK]
        --
        invoice_id : uuid [FK]
        product_id : uuid [FK]
        description : text
        quantity : decimal
        unit_price : decimal
        vat_code_id : uuid [FK]
        ledger_account_id : uuid [FK]
    }
}

package "Thema 5: Accounting & Matching" #F3E5F5 {

    entity "journals" as jour {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        name : varchar
        type : varchar
        code : varchar <<UQ>>
    }

    entity "ledger_accounts" as ledger {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        code : integer <<UQ>>
        name : varchar
        category : varchar
    }

    entity "journal_entries" as entries {
        * id : uuid [PK]
        --
        journal_id : uuid [FK]
        date : date
        description : text
    }

    entity "transactions" as trans {
        * id : uuid [PK]
        --
        entry_id : uuid [FK]
        ledger_id : uuid [FK]
        debit : decimal
        credit : decimal
        is_reconciled : boolean
    }

    entity "match_p5_p1" as matchPurchase {
        * id : uuid [PK]
        --
        outgoing_order_id : uuid [FK]
        incoming_invoice_id : uuid [FK]
        journal_entry_id : uuid [FK] (opt)
        match_confidence : decimal
    }

    entity "match_p2_p4" as matchSales {
        * id : uuid [PK]
        --
        incoming_order_id : uuid [FK]
        outgoing_invoice_id : uuid [FK]
        journal_entry_id : uuid [FK] (opt)
        match_confidence : decimal
    }

    entity "reconciliation_log" as recon {
        * id : uuid [PK]
        --
        bank_statement_id : uuid [FK]
        journal_entry_id : uuid [FK]
        transaction_id : uuid [FK]
        match_type : varchar
    }

    entity "assets" as assets {
        * id : uuid [PK]
        --
        purchase_entry_id : uuid [FK]
        ledger_account_id : uuid [FK]
        total_value : decimal
        depreciation_months : integer
    }

    entity "budgets" as budgets {
        * id : uuid [PK]
        --
        ledger_code : uuid [FK]
        amount_limit : decimal
        period : varchar
    }
}

package "Thema 6: Governance & Logs" #ECEFF1 {

    entity "accounting_periods" as periods {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        name : varchar(20)
        start_date : date
        end_date : date
        is_locked : boolean
        locked_by : uuid [FK]
    }

    entity "audit_logs" as audit {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        user_id : uuid [FK]
        action : varchar(100)
        entity_type : varchar(50)
        entity_id : uuid
        changes : jsonb
        timestamp : timestamp
    }

    entity "follow_up_logs" as follow {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        match_p5_p1_id : uuid [FK]
        match_p2_p4_id : uuid [FK]
        user_id : uuid [FK]
        action_taken : text
        status_after_action : varchar(50)
    }

    entity "interaction_logs" as interact {
        * id : uuid [PK]
        --
        org_id : uuid [FK]
        partner_id : uuid [FK]
        entity_type : varchar(50)
        entity_id : uuid
        type : varchar(20)
        message : text
    }

    entity "invoice_disputes" as disputes {
        * id : uuid [PK]
        --
        incoming_invoice_id : uuid [FK]
        outgoing_invoice_id : uuid [FK]
        reason : text
        status : varchar(20)
    }

    entity "user_dashboard_configs" as dash {
        * id : uuid [PK]
        --
        user_id : uuid [FK]
        layout : jsonb
        theme : varchar(20)
    }
}
